let chatHistory = [];


function startChat() {

  if (!checkApiKey()) return;

  document.getElementById('modeSelect').style.display = 'none';

  document.getElementById('chatScreen').classList.add('active');

  chatHistory = [];

  document.getElementById('chatMessages').innerHTML = '';

function handleEnter(e) {

  if (e.key === 'Enter' && !e.shiftKey) {

    e.preventDefault();

    sendChat();

  }

}


function sendChat() {

  const input = document.getElementById('chatInput');

  const text = input.value.trim();

  if (!text) return;

  if (!checkApiKey()) return;


  appendMessage('user', text);

  chatHistory.push({ role: 'user', content: text });

  input.value = '';


  sendToGAS('chat', chatHistory, false);

}


function continueChat() {

  document.getElementById('chatSettei').classList.remove('active');

  document.getElementById('chatInputArea').style.display = '';

}


function requestSummary() {

  if (chatHistory.length === 0) {

    alert('まず少し話してみてください。');

    return;

  }

  const summaryMsg = [

    { role: 'user', content: '今までの会話をもとに設定書にまとめてください' }

  ];

  sendToGAS('summary', [...chatHistory, ...summaryMsg], false, true);

}


async function sendToGAS(mode, messages, isFirst, isSummary) {

  setLoading(true, 'chat');


  // 最初のメッセージは空のユーザーメッセージで始める

  let msgs = messages;

  if (isFirst) {

    msgs = [{ role: 'user', content: '始めてください' }];

  }


  try {

    const res = await fetch(GAS_URL, {

      method: 'POST',

      headers: { 'Content-Type': 'application/json' },

      body: JSON.stringify({ mode, messages: msgs, apiKey })

    });

    const data = await res.json();


    if (data.error) {

      appendMessage('ai', 'エラーが発生しました：' + data.error);

    } else {

      const reply = data.reply;


      if (isSummary || reply.includes('---設定書ここから---')) {

        // 設定書を表示

        showSettei(reply, 'chat');

      } else {

        appendMessage('ai', reply);

        chatHistory.push({ role: 'assistant', content: reply });

      }

    }

  } catch (err) {

    appendMessage('ai', '通信エラーが発生しました。GAS URLを確認してください。');

  }


  setLoading(false, 'chat');

}


function appendMessage(role, text) {

  const container = document.getElementById('chatMessages');

  const div = document.createElement('div');

  div.className = 'message ' + role;


  const label = document.createElement('div');

  label.className = 'message-label';

  label.textContent = role === 'ai' ? 'AI' : 'あなた';


  const bubble = document.createElement('div');

  bubble.className = 'message-bubble';

  bubble.textContent = text;


  div.appendChild(label);

  div.appendChild(bubble);

  container.appendChild(div);

  container.scrollTop = container.scrollHeight;

}

function copySettei(id) {

  const text = document.getElementById(id).textContent;

  navigator.clipboard.writeText(text).then(() => {

    const btn = event.target;

    const orig = btn.textContent;

    btn.textContent = 'コピーしました！';

    setTimeout(() => { btn.textContent = orig; }, 2000);

  }).catch(() => {

    // フォールバック

    const ta = document.createElement('textarea');

    ta.value = text;

    document.body.appendChild(ta);

    ta.select();

    document.execCommand('copy');

    document.body.removeChild(ta);

    const btn = event.target;

    const orig = btn.textContent;

    btn.textContent = 'コピーしました！';

    setTimeout(() => { btn.textContent = orig; }, 2000);

  });

}
